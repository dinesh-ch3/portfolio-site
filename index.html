-- Hospital Admissions & Capacity Intelligence
WITH AdmissionMetrics AS (
  SELECT 
    department,
    COUNT(patient_id) AS total_admissions,
    ROUND(AVG(discharge_date - admission_date), 1) AS avg_stay_days,
    DENSE_RANK() OVER(ORDER BY COUNT(patient_id) DESC) AS dept_rank
  FROM healthcare_records
  WHERE discharge_date IS NOT NULL
  GROUP BY department
)
SELECT department, total_admissions, avg_stay_days
FROM AdmissionMetrics
WHERE dept_rank <= 5;
